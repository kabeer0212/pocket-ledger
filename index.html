Apps Script doGet setup helper (Click to view)

To let Pocket Ledger fetch your custom categories live, make sure your Apps Script has this doGet(e) handler deployed with access set to Anyone:

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var catSheet = ss.getSheetByName('categories') || ss.getSheetByName('Categories');
  var categories = [];
  if (catSheet) {
    var vals = catSheet.getDataRange().getValues();
    categories = vals.map(function(r) { return String(r[0] || '').trim(); }).filter(Boolean);
  }
  if (categories.length === 0) {
    var rec = ss.getSheetByName('record');
    if (rec && rec.getLastRow() >= 8) {
      var d = rec.getRange(8, 4, rec.getLastRow() - 7, 1).getValues();
      d.forEach(function(r) {
        var c = String(r[0] || '').trim();
        if (c && categories.indexOf(c) === -1) categories.push(c);
      });
    }
  }
  if (categories.length === 0) {
    categories = ['Food', 'Transportation', 'Textbooks', 'Accessories', 'Fun', 'Rent', 'Miscellaneous', 'Events & Outings'];
  }
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    categories: categories
  })).setMimeType(ContentService.MimeType.JSON);
}
